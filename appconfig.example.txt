# PLCSIM-AutoStart settings (template).
# The installer copies this file to appconfig.txt next to PlcsimAutoStart.exe.
# The tool rewrites appconfig.txt when you change a setting on the web page,
# so edit it by hand only while the service is STOPPED.
# An empty value means "use the default".
# Every setting is explained in docs/CONFIGURATION.md.

# ---- Web server ----
# Address the page listens on. http://+:8090/ = reachable from the network (the installer also opens
# the port in the firewall). Use http://localhost:8090/ to allow only this machine.
http_prefix = http://+:8090/

# ---- Workspace ----
# Folder that holds your PLCSIM workspaces. Empty = <Documents>\PLCSIM of the account the app runs as.
workspace_root =
# The workspace the page shows. Empty = the most recently changed one.
workspace =

# ---- Auto-start at boot ----
autostart_enabled = false
# last  = power on the PLCs that were on before the restart.
# fixed = always power on the PLC named in autostart_instance.
autostart_mode = last
autostart_instance =

# ---- How many PLCs can be on ----
# The most PLCs the web page lets you power on at once. It can be higher than the auto-start cap.
# Only raise it if this machine can run that many PLCs at once.
max_powered_on = 1
# The auto-start cap: the most PLCs auto-start powers on at boot.
# Set it to the number of PLCs this machine can really run.
hard_max_powered_on = 1

# ---- Boot protections (loop-breaker) ----
# How many boots in a row can fail the check below before auto-start is skipped (SAFE MODE).
boot_fail_limit = 2
# After auto-start, how long (seconds) PLCSIM and the page must keep answering for the boot to count as good.
stable_seconds = 90
# Pause (ms) between PLCs during auto-start, so they don't all start at the same moment.
start_stagger_ms = 4000
# How often (ms) the tool checks that its own page answers, and how long (ms) a check may take
# before it counts as failed.
health_probe_interval_ms = 15000
health_timeout_ms = 3000
# Used by the tool; leave it empty.
last_running =

# ---- Network ----
# Softbus = PLCSIM's simulation network, no adapter (default).
# TCPIPSingleAdapter / TCPIPMultipleAdapter = TCP/IP through the machine's network.
network_mode = Softbus
# Adapter ifIndex and name. Only used by TCPIPMultipleAdapter; pick it on the web page.
adapter_index = 0
adapter_name =

# ---- Advanced ----
# "default" = PLCSIM's own saved storage, so each PLC keeps its program after a reboot.
# + New PLC and Delete need this value.
storage_layout = default
# How long (ms) to wait for a power-on / power-off, and for RUN / STOP.
poweron_timeout = 60000
run_timeout = 60000
# At startup, how long (seconds) to wait for PLCSIM to answer.
connect_wait_seconds = 120
# Full path of Siemens.Simatic.Simulation.Runtime.Api.x64.dll. Empty = find the installed PLCSIM Advanced.
api_dll_path =
# Log file path. Empty = autostart.log next to PlcsimAutoStart.exe.
log =
